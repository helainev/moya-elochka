// Функция автоматической загрузки игрушек из базы данных GitHub async function loadDynamicCatalog() { const container = document.getElementById('catalog-container'); // Ссылка на файл базы данных в вашей ветке develop (или main после мерджа) const url = 'https://githubusercontent.com'; try { const res = await fetch(url); if (!res.ok) throw new Error('Файл базы данных еще не создан'); const toys = await res.json(); toys.forEach(toy => { const card = document.createElement('div'); card.className = "bg-slate-800/80 backdrop-blur rounded-2xl p-4 shadow-2xl text-center flex flex-col justify-between"; // Проверяем статус для правильного цвета баджа const isAvailable = toy.status === 'instock'; const statusBadge = isAvailable ? ' В наличии' : ' Под заказ'; card.innerHTML = `

${toy.name}

${toy.price}

${statusBadge} ${isAvailable ? 'Купить' : 'Написать мастеру'}
`; container.appendChild(card); }); } catch (error) { console.log("Динамический каталог пуст или загружается локально:", error.message); // Здесь можно оставить ваши 2 старые ручные карточки как заглушку, если файла базы данных на GitHub еще нет } } // Запускаем чтение при загрузке сайта loadDynamicCatalog();
